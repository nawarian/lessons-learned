Step 1: 📥 Collect the Data

Goal: Download ~1B tokens of mixed text to data/raw/ as plain UTF-8 text files.

Each dataset has a different schema. We normalize everything to plain text — one line per document/session. This is the only script that needs internet access.

01_collect_data.py
"""
Download and normalize all data sources into data/raw/.
Usage: python 01_collect_data.py
Disk needed: ~2-3 GB for raw text
"""
import os, json
from datasets import load_dataset
from tqdm import tqdm

os.makedirs("data/raw", exist_ok=True)

# ── 1. FineWeb-Edu (English, 400M tokens ≈ 50K samples) ──
print("[1/4] Downloading FineWeb-Edu (English)...")
en = load_dataset("HuggingFaceFW/fineweb-edu", split="train", streaming=True)
with open("data/raw/en.txt", "w", encoding="utf-8") as f:
    for i, row in tqdm(en.shuffle(seed=42).take(50_000), desc="FineWeb"):
        f.write(row["text"].strip() + "\n")

# ── 2. CulturaX (Portuguese, 250M tokens ≈ 30K samples) ──
print("[2/4] Downloading CulturaX (Portuguese)...")
cx = load_dataset("uonlp/CulturaX", split="train", streaming=True)
pt = cx.filter(lambda x: x["language"] == "pt")
with open("data/raw/pt.txt", "w", encoding="utf-8") as f:
    for i, row in tqdm(pt.shuffle(seed=42).take(30_000), desc="CulturaX"):
        f.write(row["text"].strip() + "\n")

# ── 3. The Stack (Code, 150M tokens ≈ 15K samples) ──
print("[3/4] Downloading The Stack (Code)...")
stack = load_dataset("bigcode/the-stack-dedup", split="train", streaming=True)
langs = ["python", "typescript", "javascript", "rust", "go", "java", "shell", "lua"]
code = stack.filter(lambda x: x.get("lang", "") in langs)
with open("data/raw/code.txt", "w", encoding="utf-8") as f:
    for i, row in tqdm(code.shuffle(seed=42).take(15_000), desc="TheStack"):
        f.write(row["content"].strip() + "\n")

# ── 4. Agentic Traces (200M tokens, full dataset) ──
print("[4/4] Downloading Agentic Traces...")
agent = load_dataset("MaxDevv/real-pi-coding-agent-traces-sessions", split="train")
with open("data/raw/agent.txt", "w", encoding="utf-8") as f:
    for session in tqdm(agent, desc="Agent"):
        # Each session is a list of JSON events (pi format)
        # Reconstruct the conversation as plain text
        lines = []
        for event in session["events"]:
            if event["type"] == "message":
                role = event["message"]["role"]
                for part in event["message"]["content"]:
                    if part["type"] == "text":
                        lines.append(f"<|{role}|> {part['text']}")
                    elif part["type"] in ("bash", "read", "write", "edit"):
                        cmd = part.get("command") or part.get("path") or ""
                        lines.append(f"<|tool|> {part['type']}: {cmd}")
        f.write("\n".join(lines) + "\n\n---END OF SESSION---\n\n")

print("Done! Files:")
for fn in os.listdir("data/raw"):
    size = os.path.getsize(f"data/raw/{fn}") / 1e6
    print(f"  {fn}: {size:.1f} MB")
Run time: ~1-2 hours (download speed dependent). Most of the time is spent on The Stack and CulturaX filtering — the streaming API needs to scan shards to find matching records. FineWeb-Edu and Agentic traces are faster since they don't need language filtering.
Token count note: "50K samples" is a rough heuristic. FineWeb-Edu samples average ~8K tokens each (they're full web pages). CulturaX samples average ~8-10K tokens. The exact token count depends on the tokenizer. After tokenizer training in Step 2, verify with wc -c data/raw/*.txt and adjust take(N) if needed.

Step 2: 🏗️ Train the BPE Tokenizer

Goal: Train a 32,768-vocabulary BPE tokenizer on all four data sources combined. This runs locally on your GPU machine (low compute).

02_train_tokenizer.py
"""
Train a BPE tokenizer on the mixed corpus.
Usage: python 02_train_tokenizer.py
Output: data/tokenizer.json (~4 MB)
"""
from tokenizers import Tokenizer, models, trainers, pre_tokenizers, processors

files = ["data/raw/en.txt", "data/raw/pt.txt",
         "data/raw/code.txt", "data/raw/agent.txt"]

tokenizer = Tokenizer(models.BPE())
tokenizer.pre_tokenizer = pre_tokenizers.ByteLevel(add_prefix_space=False)

trainer = trainers.BpeTrainer(
    vocab_size=32_768,
    min_frequency=2,
    special_tokens=["[PAD]", "[MASK]", "[UNK]", "[BOS]", "[EOS]"],
    show_progress=True,
)

tokenizer.train(files, trainer)

# Add post-processor: wrap in [BOS]...[EOS] tokens
tokenizer.post_processor = processors.TemplateProcessing(
    single="[BOS] $A [EOS]",
    pair="[BOS] $A [EOS] $B:1 [EOS]:1",
    special_tokens=[
        ("[BOS]", tokenizer.token_to_id("[BOS]")),
        ("[EOS]", tokenizer.token_to_id("[EOS]")),
    ],
)

tokenizer.save("data/tokenizer.json")
print(f"Vocab size: {tokenizer.get_vocab_size()}")
print(f"Saved to data/tokenizer.json")

# Test it
test_texts = [
    "Hello, world! This is a test in English.",
    "Olá, mundo! Este é um teste em Português.",
    "def hello(name: str) -> str:\n    return f'Hello, {name}'",
]
for t in test_texts:
    enc = tokenizer.encode(t)
    dec = tokenizer.decode(enc.ids)
    print(f"\nInput:  {t[:60]}")
    print(f"Tokens: {enc.tokens[:20]}... ({len(enc.ids)} tokens)")
    print(f"Roundtrip: {'✅' if dec == t else '❌'}")
Why ByteLevel pre-tokenizer? ByteLevel handles all Unicode (Portuguese accents, code symbols) without needing a separate normalization step. It also ensures that any byte sequence can be represented — no out-of-vocabulary tokens for code.

Step 3: 📦 Tokenize and Shard

Goal: Convert raw text to binary .bin shards — uint16 token IDs. This format is fast to load during training (no parsing, just mmap).

03_tokenize_shard.py
"""
Tokenize all raw text files and save as binary shards.
Usage: python 03_tokenize_shard.py
Output: data/train/*.bin (uint16 token IDs)

Each shard is a flat array of uint16 values:
  [n_tokens] followed by the token IDs.
Header-free — just raw uint16s.
"""
import os, json, mmap
import numpy as np
from tokenizers import Tokenizer
from tqdm import tqdm

tokenizer = Tokenizer.from_file("data/tokenizer.json")
os.makedirs("data/train", exist_ok=True)

TOKENS_PER_SHARD = 50_000_000  # 50M tokens per shard (~100 MB per shard)
MASK_ID = tokenizer.token_to_id("[MASK]")

input_files = ["data/raw/en.txt", "data/raw/pt.txt",
               "data/raw/code.txt", "data/raw/agent.txt"]

all_tokens = []
for fn in input_files:
    if not os.path.exists(fn):
        print(f"Skipping {fn} (not found)")
        continue
    print(f"Tokenizing {fn}...")
    with open(fn, "r", encoding="utf-8") as f:
        for line in tqdm(f, desc=os.path.basename(fn)):
            line = line.strip()
            if not line:
                continue
            ids = tokenizer.encode(line).ids
            all_tokens.extend(ids)

    # Flush shards periodically to avoid OOM
    while len(all_tokens) >= TOKENS_PER_SHARD:
        shard = all_tokens[:TOKENS_PER_SHARD]
        all_tokens = all_tokens[TOKENS_PER_SHARD:]
        shard_id = len(os.listdir("data/train"))
        arr = np.array(shard, dtype=np.uint16)
        arr.tofile(f"data/train/shard_{shard_id:04d}.bin")

# Flush remaining
if all_tokens:
    shard_id = len(os.listdir("data/train"))
    arr = np.array(all_tokens, dtype=np.uint16)
    arr.tofile(f"data/train/shard_{shard_id:04d}.bin")

print(f"\nDone! Created {len([f for f in os.listdir('data/train') if f.endswith('.bin')])} shards")
total = sum(os.path.getsize(f"data/train/{f}") for f in os.listdir("data/train") if f.endswith(".bin"))
print(f"Total tokens: {total // 2:,} ({(total / 1e9):.2f} GB)")

Expected output: ~20 shards × 50M tokens = ~1B tokens, ~2 GB on disk.

4a. Configuration

config.py
"""
Hyperparameters for the 300M MDLM student.
"""
from dataclasses import dataclass, field
from typing import Optional

@dataclass
class MDLMConfig:
    # Architecture
    vocab_size: int = 32_768
    d_model: int = 1024
    n_layers: int = 20
    n_heads: int = 16
    d_ff: int = 4_096          # 4× d_model
    max_seq_len: int = 8_192
    dropout: float = 0.0
    bias: bool = False         # No bias in linear layers (modern practice)

    # Position encoding
    rope_theta: float = 10_000.0

    # Time conditioning
    time_dim: int = 256        # Sinusoidal → MLP hidden dim

    # Noise schedule
    schedule_type: str = "cosine"  # 'cosine' or 'linear'

    # Training
    batch_size: int = 2        # Per GPU (adjust based on VRAM)
    grad_accum_steps: int = 8  # Effective batch = 2 × 8 = 16
    learning_rate: float = 3e-4
    lr_warmup_steps: int = 1_000
    max_steps: int = 50_000
    optimizer: str = "adafactor"  # Adafactor to save VRAM

    # Distillation
    distill_alpha: float = 0.5    # Weight for distillation loss
    distill_alpha_decay: bool = True  # Decay α → 0 over training (pure diffusion at end)
    teacher_kl_temperature: float = 2.0  # Soften teacher logits

    # Paths
    tokenizer_path: str = "data/tokenizer.json"
    data_dir: str = "data/train"
    teacher_name: str = "ML-GSAI/LLaDA-8B"
    checkpoint_dir: str = "checkpoints/student"

    @property
    def total_params(self) -> int:
        """Rough parameter count."""
        emb = self.vocab_size * self.d_model
        head = self.d_model * self.vocab_size
        per_layer = 4 * self.d_model**2 + 2 * self.d_model * self.d_ff
        time_mlp = 2 * self.time_dim * self.d_model
        return emb + self.n_layers * per_layer + head + time_mlp

4b. Model

model.py
"""
300M Masked Diffusion Language Model (student).
Same architecture as your tiny MDLM from Lesson 0005, scaled up:
  - RoPE position encoding (instead of learned absolute)
  - adaLN time conditioning
  - Pre-norm transformer encoder
  - Cosine noise schedule

Usage:
    from model import MDLM, MDLMConfig
    model = MDLM(config)
"""
import math
import torch
import torch.nn as nn
import torch.nn.functional as F
from torchtyping import TensorType

# ── Rotary Position Embedding ──

class RotaryEmbedding(nn.Module):
    """RoPE from Lesson 0006. No learned params."""
    def __init__(self, dim: int, theta: float = 10_000.0):
        super().__init__()
        inv_freq = 1.0 / (theta ** (torch.arange(0, dim, 2).float() / dim))
        self.register_buffer("inv_freq", inv_freq)

    def forward(self, seq_len: int, device: torch.device):
        t = torch.arange(seq_len, device=device)
        freqs = torch.einsum("i,j->ij", t, self.inv_freq)
        return torch.cat([freqs.sin(), freqs.cos()], dim=-1)  # [L, d]

def apply_rotary(x: TensorType["B", "L", "n_h", "d_h"],
                 cos: TensorType["L", "d_h"], sin: TensorType["L", "d_h"]):
    """Rotate queries/keys by RoPE."""
    d_h = x.shape[-1]
    x1, x2 = x[..., :d_h//2], x[..., d_h//2:]
    cos = cos[:x.shape[1], None, :]
    sin = sin[:x.shape[1], None, :]
    return torch.cat([x1 * cos - x2 * sin, x1 * sin + x2 * cos], dim=-1)


# ── Time Embedding ──

class TimeEmbedding(nn.Module):
    """Sinusoidal embedding → MLP → scale + shift vectors for adaLN."""
    def __init__(self, d_model: int, time_dim: int = 256):
        super().__init__()
        self.time_dim = time_dim
        # MLP: time_dim → d_model × 2 (scale + shift per block)
        self.net = nn.Sequential(
            nn.Linear(time_dim, d_model * 4),
            nn.SiLU(),
            nn.Linear(d_model * 4, d_model * 2),
        )

    def forward(self, t: TensorType["B"]) -> TensorType["B", "d_model*2"]:
        # t: [B] — noise level in [0, 1]
        # Sinusoidal encoding
        half_dim = self.time_dim // 2
        emb = math.log(10_000) / (half_dim - 1)
        emb = torch.exp(torch.arange(half_dim, device=t.device) * -emb)
        s = t.unsqueeze(-1) * emb.unsqueeze(0)  # [B, half_dim]
        emb = torch.cat([s.sin(), s.cos()], dim=-1)  # [B, time_dim]
        return self.net(emb)  # [B, d_model * 2]


# ── AdaLN Modulation ──

class AdaLN(nn.Module):
    """Adaptive LayerNorm modulated by time embedding."""
    def __init__(self, d_model: int):
        super().__init__()
        self.norm = nn.LayerNorm(d_model, elementwise_affine=False)

    def forward(self, x: TensorType["B", "L", "d"],
                scale_shift: TensorType["B", "d*2"]):
        scale, shift = scale_shift.chunk(2, dim=-1)  # [B, d], [B, d]
        x = self.norm(x)
        return x * (1 + scale.unsqueeze(1)) + shift.unsqueeze(1)


# ── Transformer Block ──

class TransformerBlock(nn.Module):
    """Pre-norm encoder block with adaLN modulation."""
    def __init__(self, d_model: int, n_heads: int, d_ff: int, dropout: float = 0.0):
        super().__init__()
        self.ada_attn = AdaLN(d_model)
        self.attn = nn.MultiheadAttention(
            d_model, n_heads, dropout=dropout, batch_first=True, bias=False)
        self.ada_ffn = AdaLN(d_model)
        self.ffn = nn.Sequential(
            nn.Linear(d_model, d_ff, bias=False),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(d_ff, d_model, bias=False),
        )
        self.dropout = nn.Dropout(dropout)

    def forward(self, x, scale_shift, attn_mask=None):
        # Attention with adaLN
        x = x + self.dropout(self.attn(
            self.ada_attn(x, scale_shift),
            self.ada_attn(x, scale_shift),
            self.ada_attn(x, scale_shift),
            attn_mask=attn_mask, need_weights=False)[0])
        # FFN with adaLN
        x = x + self.dropout(self.ffn(self.ada_ffn(x, scale_shift)))
        return x


# ── Noise Schedules ──

def cosine_schedule(t: TensorType["B"]) -> TensorType["B"]:
    """Cosine noise schedule: α(t) = cos(t * π/2)."""
    return (t * math.pi / 2).cos()

def linear_schedule(t: TensorType["B"]) -> TensorType["B"]:
    """Linear noise schedule: α(t) = 1 - t."""
    return 1 - t


# ── Full Model ──

class MDLM(nn.Module):
    """Masked Diffusion Language Model — student architecture."""
    def __init__(self, config):
        super().__init__()
        self.config = config

        # Token embedding
        self.token_embed = nn.Embedding(config.vocab_size, config.d_model)

        # RoPE (one shared instance, not per-layer)
        self.rotary = RotaryEmbedding(config.d_model // config.n_heads, config.rope_theta)

        # Time embedding (one shared instance)
        self.time_embed = TimeEmbedding(config.d_model, config.time_dim)

        # Transformer blocks
        self.blocks = nn.ModuleList([
            TransformerBlock(config.d_model, config.n_heads, config.d_ff, config.dropout)
            for _ in range(config.n_layers)
        ])

        # Final norm + head
        self.final_norm = nn.LayerNorm(config.d_model)
        self.head = nn.Linear(config.d_model, config.vocab_size, bias=False)

        # Cache time embedding projections to avoid redundant computation
        self._scale_shift_cache = {}

        # Initialize weights
        self.apply(self._init_weights)

    def _init_weights(self, module):
        if isinstance(module, nn.Linear):
            nn.init.xavier_uniform_(module.weight)
        elif isinstance(module, nn.Embedding):
            nn.init.normal_(module.weight, std=0.02)

    def get_schedule(self, t: TensorType["B"]) -> TensorType["B"]:
        """Get α(t) for the configured schedule."""
        if self.config.schedule_type == "cosine":
            return cosine_schedule(t)
        return linear_schedule(t)

    def get_mask(self, t: TensorType["B"], seq_len: int) -> TensorType["B", "L"]:
        """Sample a binary mask given noise level t.
        
        Each sample gets α(t) * seq_len tokens masked (randomly selected).
        Returns: [B, L] — True = masked.
        """
        B = t.shape[0]
        alpha = self.get_schedule(t)  # [B]
        n_mask = (alpha * seq_len).long().clamp(1, seq_len - 1)

        # Randomly select which positions to mask
        rand = torch.rand(B, seq_len, device=t.device)
        # Mask the n_mask[i] positions with lowest rand values
        threshold = rand.sort(dim=-1).values[torch.arange(B), n_mask - 1]
        mask = rand <= threshold.unsqueeze(-1)
        return mask

    def forward(self, x: TensorType["B", "L"],
                t: TensorType["B"],
                mask: TensorType["B", "L"] = None) -> TensorType["B", "L", "V"]:
        """Forward pass: embed, apply mask, run blocks, predict logits.
        
        Args:
            x: Token IDs [B, L]. Already has [MASK] tokens applied.
            t: Noise levels [B] in [0, 1].
            mask: Boolean mask [B, L] — True = masked position.
            
        Returns:
            logits: [B, L, vocab_size] — predictions for all positions.
        """
        B, L = x.shape
        device = x.device

        # Embed tokens
        h = self.token_embed(x)  # [B, L, d_model]

        # RoPE sin/cos
        cos_sin = self.rotary(L, device)
        cos, sin = cos_sin.chunk(2, dim=-1)
        # Expand for multi-head: [L, d_h]
        d_h = self.config.d_model // self.config.n_heads
        cos = cos[:, :d_h]
        sin = sin[:, :d_h]

        # Time embedding → scale/shift per block
        scale_shift = self.time_embed(t)  # [B, d_model*2]

        # Run blocks (pre-compute scale_shift once, broadcast to all blocks)
        for block in self.blocks:
            h = block(h, scale_shift)

        h = self.final_norm(h)
        logits = self.head(h)  # [B, L, vocab_size]
        return logits

    @torch.no_grad()
    def generate(self, tokenizer, prompt_ids=None, length=512,
                 steps=64, temp=1.0, top_k=50):
        """Generate text by iterative denoising.
        
        Args:
            tokenizer: BPE tokenizer with [MASK] token.
            prompt_ids: [1, prompt_len] — optional prompt tokens.
            length: Total sequence length to generate.
            steps: Number of denoising steps.
            temp: Sampling temperature.
            top_k: Top-k filtering.
            
        Returns:
            Token IDs [1, length].
        """
        self.eval()
        device = next(self.parameters()).device
        mask_id = tokenizer.token_to_id("[MASK]")

        if prompt_ids is not None:
            prompt_len = prompt_ids.shape[1]
            x = torch.full((1, length), mask_id, dtype=torch.long, device=device)
            x[:, :prompt_len] = prompt_ids
        else:
            x = torch.full((1, length), mask_id, dtype=torch.long, device=device)

        t_vals = torch.linspace(1.0, 0.0, steps, device=device)
        for step, t in enumerate(t_vals):
            # Current mask: where x == [MASK]
            mask = (x == mask_id)

            # Predict logits for all positions
            logits = self.forward(x, t.unsqueeze(0), mask)
            logits = logits / temp

            # Only consider masked positions
            logits_masked = logits[0, mask[0]]

            # Top-k filtering
            if top_k > 0:
                top_vals, _ = torch.topk(logits_masked, top_k, dim=-1)
                logits_masked[logits_masked < top_vals[:, -1:]] = -float('inf')

            # Sample
            probs = F.softmax(logits_masked, dim=-1)
            samples = torch.multinomial(probs, 1).squeeze(-1)

            # Compute confidence of predictions
            confidence = probs.max(dim=-1).values

            # Determine how many tokens to reveal at this step
            n_unmasked = mask.sum()
            n_to_reveal = max(1, n_unmasked // (steps - step))

            # Reveal the highest-confidence predictions
            _, indices = confidence.topk(n_to_reveal)
            x[0, mask[0].nonzero().squeeze(-1)[indices]] = samples[indices]

        return x
Key differences from your tiny MDLM (Lesson 0005):
  • RoPE instead of learned absolute positions — can extrapolate to longer sequences
  • Pre-norm architecture (LayerNorm before each sublayer, not after) — more stable at 20 layers
  • Per-layer adaLN — each block receives the time modulation independently
  • Flash-Attention compatible — uses nn.MultiheadAttention (swap to Flash for speed)

Goal: Load LLaDA 8B in BF16 and verify it runs on your L40S (48 GB). This model generates the soft targets for distillation.

05_setup_teacher.py
"""
Load LLaDA 8B in BF16 and verify inference works.
Usage: python 05_setup_teacher.py
VRAM: ~16 GB (BF16)
"""
import torch
from transformers import AutoModelForCausalLM, AutoTokenizer

model_name = "ML-GSAI/LLaDA-8B"

print(f"Loading LLaDA 8B in BF16...")
model = AutoModelForCausalLM.from_pretrained(
    model_name,
    torch_dtype=torch.bfloat16,
    device_map="auto",
    trust_remote_code=True,

Step 6: 🔬 Distill — Train the Student

Goal: Train the 300M student to match LLaDA 8B's logits at multiple noise levels. This is the main training script — runs for ~1.2 days on L40S.

06_distill.py
"""
Distill LLaDA 8B → 300M MDLM student.
Usage: python 06_distill.py

Training loop:
1. Sample a batch of token sequences from shards
2. Randomly mask a subset of tokens (noise level t)
3. Forward pass through student (predict masked tokens)
4. Forward pass through teacher (BF16, no grad) — soft targets
5. Loss = α * KL(student_logits || teacher_logits) + (1-α) * L_diffusion
6. Backprop through student only
"""
import os, math, time
import torch
import torch.nn.functional as F
from torch.utils.data import DataLoader, IterableDataset
import numpy as np
from transformers import AutoModel
from tokenizers import Tokenizer

from config import MDLMConfig
from model import MDLM

# ── Dataset ──

class ShardDataset(IterableDataset):
    """Streams uint16 token IDs from binary shards."""
    def __init__(self, data_dir: str, seq_len: int):
        self.shards = sorted([
            os.path.join(data_dir, f)
            for f in os.listdir(data_dir) if f.endswith(".bin")
        ])
        self.seq_len = seq_len

    def __iter__(self):
        while True:
            for shard_path in self.shards:
                tokens = np.fromfile(shard_path, dtype=np.uint16).astype(np.int64)
                # Yield random sequences
                for _ in range(len(tokens) // self.seq_len):
                    start = np.random.randint(0, len(tokens) - self.seq_len)
                    yield torch.tensor(tokens[start:start + self.seq_len])

# ── Loss Functions ──

def diffusion_loss(logits, targets, mask):
    """Cross-entropy loss on masked positions only."""
    # logits: [B, L, V], targets: [B, L], mask: [B, L]
    loss = F.cross_entropy(
        logits.view(-1, logits.size(-1)),
        targets.view(-1),
        reduction='none'
    )  # [B*L]
    return loss.view_as(mask)[mask].mean()

def distillation_kl(student_logits, teacher_logits, mask, temperature=2.0):
    """KL divergence between student and teacher distributions (masked positions only)."""
    # Soften distributions with temperature
    s = F.log_softmax(student_logits / temperature, dim=-1)
    t = F.softmax(teacher_logits / temperature, dim=-1)
    # KL = sum(t * (log t - log s)) averaged over masked positions
    kl = F.kl_div(s, t, reduction='none').sum(-1)  # [B, L]
    return kl[mask].mean() * (temperature ** 2)  # Scale back

# ── Main Training Loop ──

def train():
    config = MDLMConfig()
    device = "cuda" if torch.cuda.is_available() else "cpu"
    print(f"Device: {device}")

    # Student
    student = MDLM(config).to(device)
    print(f"Student params: {sum(p.numel() for p in student.parameters()):,}")

    # Teacher (BF16)
    print("Loading teacher (LLaDA 8B, BF16)...")
    teacher = AutoModel.from_pretrained(
        config.teacher_name,
        torch_dtype=torch.bfloat16,
        device_map="auto",
        trust_remote_code=True,
    ).eval()
    for p in teacher.parameters():
        p.requires_grad = False
    print("Teacher loaded.")

    # Tokenizer
    tokenizer = Tokenizer.from_file(config.tokenizer_path)
    mask_id = tokenizer.token_to_id("[MASK]")

    # Data
    dataset = ShardDataset(config.data_dir, config.max_seq_len)
    loader = DataLoader(dataset, batch_size=config.batch_size)

    # Optimizer (Adafactor — saves ~8 GB vs AdamW)
    from transformers import Adafactor
    optim = Adafactor(
        student.parameters(),
        lr=config.learning_rate,
        relative_step=False,
        scale_parameter=False,
        warmup_init=False,
    )

    # Training loop
    os.makedirs(config.checkpoint_dir, exist_ok=True)
    step = 0
    accum_loss = 0.0
    best_loss = float('inf')
    t_start = time.time()

    print(f"\nStarting distillation ({config.max_steps} steps)...")
    student.train()

    for batch in loader:
        if step >= config.max_steps:
            break

        # ── 1. Prepare input ──
        x = batch.to(device)  # [B, L]
        B, L = x.shape

        # Random noise level for each sample
        t = torch.rand(B, device=device)  # [B] uniform in [0,1]

        # Generate mask
        mask = student.get_mask(t, L)  # [B, L] — True = masked

        # Apply mask: replace masked positions with [MASK] token
        x_masked = x.clone()
        x_masked[mask] = mask_id

        # ── 2. Student forward ──
        student_logits = student(x_masked, t, mask)  # [B, L, V]

        # ── 3. Teacher forward (no grad) ──
        with torch.no_grad():
            # Teacher sees the same masked input
            teacher_logits = teacher(x_masked, t, mask).logits  # [B, L, V]

        # ── 4. Loss ──
        loss_diff = diffusion_loss(student_logits, x, mask)
        loss_kl = distillation_kl(
            student_logits, teacher_logits, mask,
            temperature=config.teacher_kl_temperature,
        )

        # Decay distillation weight over training
        if config.distill_alpha_decay:
            alpha = config.distill_alpha * (1 - step / config.max_steps)
        else:
            alpha = config.distill_alpha

        loss = alpha * loss_kl + (1 - alpha) * loss_diff

        # Scale by grad accum steps
        loss = loss / config.grad_accum_steps
        loss.backward()

        accum_loss += loss.item()

        # ── 5. Optimizer step ──
        if (step + 1) % config.grad_accum_steps == 0:
            torch.nn.utils.clip_grad_norm_(student.parameters(), 1.0)
            optim.step()
            optim.zero_grad()

        # ── 6. Logging ──
        if step % 100 == 0:
            elapsed = time.time() - t_start
            tok_per_sec = (step + 1) * B * L / elapsed
            print(f"Step {step:>6d}/{config.max_steps} | "
                  f"loss={accum_loss/(step+1):.4f} | "
                  f"diff={loss_diff.item():.4f} | "
                  f"kl={loss_kl.item():.4f} | "
                  f"tok/s={tok_per_sec:.0f} | "
                  f"α={alpha:.3f}")

        # ── 7. Save checkpoint ──
        if step % 5_000 == 0 and step > 0:
            ckpt = os.path.join(config.checkpoint_dir, f"student_{step:06d}.pt")
            torch.save(student.state_dict(), ckpt)
            print(f"  → Saved {ckpt}")

        step += 1

    # Final save
    ckpt = os.path.join(config.checkpoint_dir, "student_final.pt")
    torch.save(student.state_dict(), ckpt)
    elapsed = time.time() - t_start
    print(f"\nDone! {step} steps in {elapsed/3600:.1f} hours")
    print(f"Final checkpoint: {ckpt}")
    print(f"Average throughput: {step * B * L / elapsed:.0f} tok/s")

if __name__ == "__main__":
    train()

VRAM optimization tips if you're tight:
  1. Gradient checkpointing: Add student.gradient_checkpointing_enable() after creating the student — trades ~20% compute for ~40% less activation memory.
  2. Smaller batch size: Drop to batch_size=1, double grad_accum_steps to 16. Same effective batch, half the peak VRAM.
  3. Cache teacher logits offline: Run teacher once on the entire dataset, save logits, then train student without teacher in memory. Frees the full ~16 GB teacher weights from VRAM — useful if you want to train at higher batch sizes or on a smaller GPU.

7b. Quantitative — Perplexity on held-out data

# Compute perplexity on a held-out set of masked sequences
def compute_ppl(model, data_loader, tokenizer):
    """Compute perplexity on masked token prediction."""
    model.eval()
    total_loss = 0.0
    total_tokens = 0
    mask_id = tokenizer.token_to_id("[MASK]")

    with torch.no_grad():
        for batch in data_loader:
            x = batch.cuda()
            B, L = x.shape
            t = torch.rand(B, device="cuda")
            mask = model.get_mask(t, L)
            x_masked = x.clone()
            x_masked[mask] = mask_id

            logits = model(x_masked, t, mask)
            loss = F.cross_entropy(
                logits.view(-1, logits.size(-1)),
                x.view(-1),
                reduction='sum'
            )
            total_loss += loss.item()
            total_tokens += mask.sum().item()

    ppl = math.exp(total_loss / total_tokens)
    return ppl

7c. Teacher–Student Agreement

# Compare student vs teacher on a shared test set
def compute_agreement(student, teacher, data_loader, tokenizer):
    """What fraction of the time does student predict the same token as teacher?"""
    student.eval()
    teacher.eval()
    correct = 0
    total = 0

    with torch.no_grad():
        for batch in data_loader:
            x = batch.cuda()
            B, L = x.shape
            t = torch.full((B,), 0.5, device="cuda")  # Fixed noise level
            mask = student.get_mask(t, L)
            x_masked = x.clone()
            x_masked[mask] = tokenizer.token_to_id("[MASK]")

            s_logits = student(x_masked, t, mask)
            t_logits = teacher(x_masked, t, mask).logits

            s_tokens = s_logits.argmax(-1)
            t_tokens = t_logits.argmax(-1)

            correct += (s_tokens[mask] == t_tokens[mask]).sum().item()
            total += mask.sum().item()

    return correct / total
